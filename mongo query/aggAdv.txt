// for   each order   calculate  total price of items (Quantity *  price)

db.orders.aggregate([
  {
    $addFields:{
      itemsTotal:{
        $sum:{
          
        $map:{
           input:"$items",
          as:"item",
          in: {
             $multiply:["$$item.price","$$item.quantity"]
            
          }
        }
        }
      }
    }
  },
  {
    $project:{
      _id:0,
      orderId:"orderId"
      itemsTotal:1,
      }
      
    }
])

db.orders.aggregate([
  {
   $group:{
     _id:"$customerId",
     totalSpent:{
       $sum: "$totalAmount"
     }
   } 
  },{
    $addFields:{
      category:{
    $cond:{
      if: {$gt:["$totalSpent",200000]},
      then: "Premium",
      else:"Regular"
      }
    }
    }
  },{
    $merge:{
      into:"costomers",
      on:"_id",
      whenMatched:"merge",
      whenNotMatched:"discard"
    }
  }
  
])
