//find  total quantity  sold per product

db.orders.aggregate([
  {
    $unwind: {
      path: "$items",
      preserveNullAndEmptyArrays:true
    }
  },
  {
    $group: {
       _id:"$items.productName",
      totalQuantity:{
        $sum: "$items.quantity"
      }
    }
  },
  {
    $project:{
      _id:0,
      productName:"$_id",
      totalQuantity:1
    }
  }
])


=====> path →     which array to unwind
=====>>>preserveNullAndEmptyArrays → don't remove documents with empty/missing 

arrays

========>>>includeArrayIndex → add the element's original position as a new field.

//$facte  used to get multiple indights  multiple aggregation pipeline on the same input documents at same time and run result together 
like do multiple seperate calculations  and one result 

like  calculate most expensive if price >50000  and  lowproduct if price is less then 100
db.products.aggregate([
  {
    $facet:{
      expensiceProduct:[
        {
         $match:{
           price:{$gt: 50000}
         } 
        }
      ],
      lowProduct:[
        {
         $match:{
           price:{$lt:100}
         } 
        }
      ]
    }
  }
])
